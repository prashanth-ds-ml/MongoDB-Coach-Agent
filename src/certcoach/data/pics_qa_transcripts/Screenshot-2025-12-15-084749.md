A.

db.inspections.find( { result : "Pass" }).sort( {certificate_number: 1});

B.

db.inspections.find( { result : "Pass" }).sort( {certificate_number: - 1});

C.

db.inspections.find( { result : "Pass" }, { sort: { certificate_number: 1 }})

D.

db.inspections.find( { result : "Pass" }, { sort: { certificate_number: - 1 }})
